#include <iostream>
using namespace std;

int main()
{
   int n=8;//表示8*8的方格

   // c[i][j]:在位置i,j的书，由于最多为9个，所以直接用数组
   int c[10][10]={};
   c[2][3]=13;c[2][6]=6;c[3][5]=7;c[4][4]=14;c[5][2]=21;
   c[5][6]=4;c[6][3]=15;c[7][2]=14;



   // dp[i][j]：在位置i,j能获得多少书
   //只能向下(上一个位置是向上)，向右（上一个位置是左边）
   // dp[i][j]=max(dp[i-1][j],dp[i][j-1])+当前位置的书,
   //走两次=两个人走一次,则用dp[i][j][k][l]
   //k和l代表第二个人的位置
   int dp[10][10][10][10]={};
   dp[1][1][1][1]=c[1][1];//设置起始位置的书

   for(int i=1;i<=n;i++)
   {
      for(int j=1;j<=n;j++)
      {
         for(int k=1;k<=n;k++)
         {
            for(int l=1;l<=n;l++)
            {
               if(i==1&&j==1&&k==1&&l==1)
                  continue;//起始位置不用判断

               //一、1上2上（1和2两个人都从上面下来）
               int A=dp[i-1][j][k-1][l];

               //二、1上2左
               int B=dp[i-1][j][k][l-1];

               //三、1左2上
               int C=dp[i][j-1][k-1][l];

               //四、1上2左
               int D=dp[i][j-1][k][l-1];

               dp[i][j][k][l]=std::max({A,B,C,D})+c[i][j]+c[k][l];
               if(i==k&&j==l)
                  dp[i][j][k][l]-=c[k][l];
            }
         }
      }
   }


  cout<<endl<<dp[n][n][n][n]<<endl;

   system("pause");
   return 0;
}
