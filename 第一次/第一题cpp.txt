#include <iostream>
#include <string>

int main()
{

   //直接输入一行，然后统计
      std::string str;
   std::getline(std::cin, str);//读入一行
   int count=0;
   for(int i=0;i<str.size();i++)
      if(str[i]!=' ')
         count++;
   std::cout<<"Count="<< count << std::endl;
   
   system("pause");
   return 0;
}
