#include <iostream>
using namespace std;

int main()
{
   std::vector<int> a={4, 7, 3, 13, 11, 12, 0, 47, 34, 98};
   std::vector<int> b;//奇数
   std::vector<int> c;//偶数

   for(int i=0;i<a.size();i++)
   {
      if(a[i]%2==0)
         c.push_back(a[i]);//偶数
      else
         b.push_back(a[i]);
   }

   //奇数从大到小
   std::sort(b.begin(),b.end(),[](int i,int j)->bool{
      return i>j;
   });

   std::reverse(c.begin(),c.end());

   cout<<endl;
   for(int i=0;i<b.size();i++)
      cout<<b[i]<<" ";
   
   for(int i=0;i<c.size();i++)
      cout<<c[i]<<" ";

   system("pause");
   return 0;
}
