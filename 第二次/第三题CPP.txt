struct Student
{
   int id;
   int score;
   Student(int _id=0,int _score=0):id(_id),score(_score){}
};

bool compare(const Student& a,const Student &b)
{
    if(a.score==b.score)
         return a.id<b.id;
      return a.score>b.score;
}
