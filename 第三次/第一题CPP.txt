#include <vector>
#include <iostream>
using namespace std;



int main()
{
   int n=5;//班级人数

   //0设置为未归队
   std::vector<int> stu(5,0);

   //有一个同学未归队
   for(int i=0;i<n-1;i++)
   {
      int k=-1;
      cin>>k;//输入归队的同学编号
      stu[k-1]=1;//归队
   }

   for(int i=0;i<n;i++)
   {
      if(stu[i]==0)
      {
         cout<<endl<<i+1<<endl;//找到未归队的同学了
         break;
      }
   }

   system("pause");
   return 0;
}
